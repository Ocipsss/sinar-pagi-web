import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { productService } from '../services/productService'
import { productRepo } from '../db/repositories/productRepository'

export const useProductStore = defineStore('product', () => {
  const products = ref([])
  const loading = ref(false)
  const searchQuery = ref('')
  const selectedCategory = ref('SEMUA')

  const categories = computed(() => {
    const cats = new Set(products.value.map(p => p.category || 'UMUM'))
    return ['SEMUA', ...Array.from(cats)]
  })

  const filteredProducts = computed(() => {
    return products.value.filter(p => {
      const matchSearch = p.name.toLowerCase().includes(searchQuery.value.toLowerCase())
      const matchCategory = selectedCategory.value === 'SEMUA' || p.category === selectedCategory.value
      return matchSearch && matchCategory
    })
  })

  async function loadProducts() {
    loading.value = true
    try {
      const data = await productService.syncPublishedProducts()
      products.value = data || []
    } catch (err) {
      console.error('[ProductStore] Gagal memuat produk:', err)
      products.value = await productRepo.getPublished()
    } finally {
      loading.value = false
    }
  }

  return {
    products,
    loading,
    searchQuery,
    selectedCategory,
    categories,
    filteredProducts,
    loadProducts
  }
})

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { cartRepo } from '../db/repositories/cartRepository'

export const useCartStore = defineStore('cart', () => {
  const items = ref([])

  async function loadCart() {
    items.value = await cartRepo.getAll()
  }

  async function addToCart(product, qty = 1) {
    const plainProduct = JSON.parse(JSON.stringify(product))
    
    const existing = items.value.find(i => 
      i.productId === plainProduct.id && 
      i.packageId === (plainProduct.packageId || null) &&
      JSON.stringify(i.services || []) === JSON.stringify(plainProduct.services || [])
    )
    
    if (existing) {
      existing.qty += qty
      existing.updatedAt = new Date().toISOString()
      await cartRepo.save(existing)
    } else {
      const newItem = {
        id: crypto.randomUUID(),
        productId: plainProduct.id,
        name: plainProduct.name,
        originalName: plainProduct.originalName || plainProduct.name,
        price_sell: Number(plainProduct.price_sell) || 0,
        imageUrl: plainProduct.imageUrl || null,
        qty: qty,
        unit: plainProduct.unit || 'PCS',
        note: plainProduct.note || '',
        packageId: plainProduct.packageId || null,
        services: plainProduct.services || [],
        category: plainProduct.category || 'UMUM',
        updatedAt: new Date().toISOString()
      }
      await cartRepo.save(newItem)
    }
    await loadCart()
  }

  // FIX: pake cartId (id random), bukan productId
  async function updateQty(cartId, qty) {
    if (qty <= 0) {
      await cartRepo.removeById(cartId)
    } else {
      await cartRepo.updateQtyById(cartId, qty)
    }
    await loadCart()
  }

  async function removeFromCart(cartId) {
    await cartRepo.removeById(cartId)
    await loadCart()
  }

  async function clearCart() {
    await cartRepo.clear()
    items.value = []
  }

  const totalItems = computed(() => items.value.reduce((sum, i) => sum + i.qty, 0))
  const totalPrice = computed(() => items.value.reduce((sum, i) => sum + (i.price_sell * i.qty), 0))

  return { items, loadCart, addToCart, updateQty, removeFromCart, clearCart, totalItems, totalPrice }
})

import { collection, getDocs, query, where } from 'firebase/firestore'
import { db as firestoreDb } from './firebase.js'
import { productRepo } from '../db/repositories/productRepository.js'
import { db } from '../db/index.js'

const getServiceIds = (s) => {
  if (s.productIds && Array.isArray(s.productIds) && s.productIds.length) return s.productIds
  if (s.productId) return [s.productId]
  return []
}

export const productService = {
  async syncPublishedProducts() {
    try {
      if (!navigator.onLine) {
        return await productRepo.getPublished()
      }

      const q = query(collection(firestoreDb, 'products'), where('isPublished', '==', true))
      const snapshot = await getDocs(q)
      const remoteProducts = []
      snapshot.forEach(doc => {
        const data = doc.data()
        if (data.deleted) return
        remoteProducts.push({
          id: doc.id,
          name: data.name || '',
          category: data.category || 'UMUM',
          unit: data.unit || 'PCS',
          price_sell: Number(data.price_sell) || 0,
          price_modal: Number(data.price_modal) || 0,
          packPrice: Number(data.packPrice) || 0,
          packQty: Number(data.packQty) || 0,
          purchasePackName: data.purchasePackName || '',
          qty: Number(data.qty) || 0,
          minStock: Number(data.minStock) || 2,
          isPublished: data.isPublished ? 1 : 0,
          imageUrl: data.imageUrl || null,
          imagePublicId: data.imagePublicId || null,
          updatedAt: data.updatedAt || new Date().toISOString(),
          deleted: 0,
          synced: 1
        })
      })
      if (remoteProducts.length > 0) await productRepo.bulkSave(remoteProducts)

      // 2. Sync product_packages
      try {
        const pkgSnap = await getDocs(collection(firestoreDb, 'product_packages'))
        const remotePkgs = pkgSnap.docs.map(d => {
          const data = d.data()
          return { ...data, id: d.id, synced: 1, deleted: data.deleted ? 1 : 0 }
        })
        if (remotePkgs.length) await db.product_packages.bulkPut(remotePkgs)
      } catch (e) {
        console.warn('[ProductService] Gagal sync product_packages', e)
      }

      // 3. Sync services - FIX MULTI PRODUK
      try {
        const svcSnap = await getDocs(collection(firestoreDb, 'services'))
        const remoteSvcs = svcSnap.docs.map(d => {
          const data = d.data()
          // normalisasi biar support lama & baru
          const ids = getServiceIds(data)
          return {
            ...data,
            id: d.id,
            productId: data.productId || null,
            productIds: ids,
            category: (data.category || 'UMUM').toUpperCase(),
            synced: 1,
            deleted: data.deleted ? 1 : 0
          }
        })
        if (remoteSvcs.length) await db.services.bulkPut(remoteSvcs)
        console.log(`[ProductService] Synced ${remoteSvcs.length} services with multi-product support`)
      } catch (e) {
        console.warn('[ProductService] Gagal sync services', e)
      }

      return await productRepo.getPublished()
    } catch (error) {
      console.error('[ProductService] Error:', error)
      return await productRepo.getPublished()
    }
  }
}

const CLOUD_NAME = import.meta.env.VITE_CLOUDINARY_CLOUD_NAME || 'vezqidog'
const UPLOAD_PRESET = import.meta.env.VITE_CLOUDINARY_PRESET || 'sinarpagi-products'

/**
 * Compress via Canvas 800px JPEG 70%
 */
export function compressImage(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = (event) => {
      const img = new Image()
      img.onload = () => {
        const canvas = document.createElement('canvas')
        const maxWidth = 800
        let width = img.width
        let height = img.height
        if (width > maxWidth) {
          height = Math.round((height * maxWidth) / width)
          width = maxWidth
        }
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        ctx.drawImage(img, 0, 0, width, height)
        canvas.toBlob(
          (blob) => (blob ? resolve(blob) : reject(new Error('Gagal kompres'))),
          'image/jpeg',
          0.7
        )
      }
      img.onerror = (err) => reject(err)
      img.src = event.target.result
    }
    reader.onerror = (err) => reject(err)
    reader.readAsDataURL(file)
  })
}

/**
 * Upload ke Cloudinary - folder auto kebuat pas upload pertama
 * @param {Blob} imageBlob
 * @param {String} folder - 'sinar-pagi/products' | 'sinar-pagi/qris' | 'sinar-pagi/bukti-bayar'
 */
export async function uploadToCloudinary(imageBlob, folder = 'sinar-pagi/bukti-bayar') {
  const formData = new FormData()
  formData.append('file', imageBlob)
  formData.append('upload_preset', UPLOAD_PRESET)
  formData.append('folder', folder) // INI YANG BIKIN AUTO FOLDER

  const response = await fetch(`https://api.cloudinary.com/v1_1/${CLOUD_NAME}/image/upload`, {
    method: 'POST',
    body: formData
  })
  if (!response.ok) {
    const errorText = await response.text()
    throw new Error(`Cloudinary Upload Failed: ${errorText}`)
  }
  const data = await response.json()
  return {
    secure_url: data.secure_url,
    public_id: data.public_id
  }
}

export const waService = {
  sendOrderToWhatsApp(customer, items, total, paymentMethod, buktiUrl) {
    const phone = import.meta.env.VITE_WA_NUMBER || '6281234567890'
    
    const isDelivery = customer.deliveryType === 'delivery'
    const ongkir = customer.deliveryFee || 0

    // Build pesan pake \n biasa, nanti di-encode
    const lines = []
    lines.push(`*PESANAN BARU - WARUNG SINAR PAGI*`)
    lines.push(`━━━━━━━━━━━━━━━━━━━━━━`)
    lines.push(`👤 Nama: ${customer.name}`)
    lines.push(isDelivery ? `🚚 Opsi: Diantar (+ ${fmt(ongkir)})` : `🏪 Opsi: Ambil di Warung`)
    lines.push(`📍 ${isDelivery ? 'Alamat' : 'Catatan'}: ${customer.address}`)
    lines.push(``)
    lines.push(`*RINCIAN PESANAN:*`)

    items.forEach((item, idx) => {
      const qty = item.qty || 1
      const unitPrice = Number(item.price_sell) || 0
      const sub = unitPrice * qty

      // 1. Nama - potong biar ga kepanjangan banget di WA
      // Nama udah termasuk paket dari buildCartItem
      lines.push(`${idx + 1}. ${item.name} x${qty}`)
      
      // 2. Harga satuan + subtotal - 1 baris aja biar rapi
      if (qty > 1) {
        lines.push(`   ${qty} x ${fmt(unitPrice)} = ${fmt(sub)}`)
      } else {
        lines.push(`   ${fmt(sub)}`)
      }

      // 3. Detail jasa - cuma 1x aja, jangan dari note + services double
      // Kita ambil dari services yang ada harganya, biar jelas
      if (item.services && item.services.length > 0) {
        item.services.forEach(s => {
          // Kalo harganya 0 jangan ditampilin Rp 0
          if ((s.price || 0) > 0) {
            lines.push(`   + ${s.name}: ${fmt(s.price)}`)
          } else {
            lines.push(`   + ${s.name}`)
          }
        })
      }
    })

    lines.push(``)
    lines.push(`━━━━━━━━━━━━━━━━━━━━━━`)
    const subtotal = total - ongkir
    lines.push(`Subtotal: ${fmt(subtotal)}`)
    lines.push(ongkir > 0 ? `Ongkir: ${fmt(ongkir)}` : `Ongkir: Gratis`)
    lines.push(`*Total: ${fmt(total)}*`)
    lines.push(`━━━━━━━━━━━━━━━━━━━━━━`)
    lines.push(`💳 Bayar: ${paymentMethod}`)
    if (paymentMethod === 'QRIS' && buktiUrl) {
      lines.push(`📎 Bukti: ${buktiUrl}`)
    }
    lines.push(``)
    lines.push(`_Mohon konfirmasi ya kak_ 🙏`)

    const message = encodeURIComponent(lines.join('\n'))
    window.open(`https://wa.me/${phone}?text=${message}`, '_blank')
  }
}

export function buildCartItem(product, selectedServices = [], finalPrice = null, selectedPackage = null) {
  let finalUnit = product.unit || 'PCS'
  let displayName = product.name
  let notes = []

  if (selectedPackage) {
    if (selectedPackage.id) {
      finalUnit = selectedPackage.name || product.purchasePackName || 'PAKET'
      displayName = `${product.name} (${selectedPackage.name})`
      if (selectedPackage.qty_pcs) {
        displayName += ` ${selectedPackage.qty_pcs} ${product.unit || 'PCS'}`
      }
    } else if (typeof selectedPackage === 'object' && selectedPackage.name) {
      finalUnit = selectedPackage.name
      displayName = `${product.name} (${selectedPackage.name})`
    }
  }

  if (Array.isArray(selectedServices) && selectedServices.length > 0) {
    notes = selectedServices.map(s => s.name)
  }

  const priceToUse = finalPrice != null ? finalPrice : product.price_sell

  return {
    id: product.id,
    name: displayName,
    originalName: product.name,
    price_sell: Number(priceToUse) || 0,
    imageUrl: product.imageUrl,
    category: product.category,
    unit: finalUnit,
    note: notes.join(', '),
    // CLONE PLAIN BIAR GA DataCloneError DI INDEXEDDB
    services: JSON.parse(JSON.stringify(selectedServices || [])),
    packageId: selectedPackage?.id || null,
    qty: product.qty
  }
}

export function getServicesForProduct(product, allServices = []) {
  if (!product) return []
  const pCat = (product.category || '').toUpperCase()

  const custom = allServices.filter(s => {
    const ids = getServiceIds(s)
    return ids.length > 0 && ids.includes(product.id)
  })
  if (custom.length > 0) return custom // KHUSUS = OVERRIDE

  return allServices.filter(s => {
    const ids = getServiceIds(s)
    return ids.length === 0 && (s.category || '').toUpperCase() === pCat
  })
}

import { db } from '../index.js'

export const productRepo = {
  getPublished: async () => {
    const list = await db.products.where('isPublished').equals(1).toArray()
    return list.filter(p => !p.deleted)
  },
  getById: async (id) => {
    const item = await db.products.get(id)
    return item && item.isPublished && !item.deleted ? item : null
  },
  getByCategory: async (category) => {
    if (!category || category === 'SEMUA') return productRepo.getPublished()
    const list = await db.products.where('category').equals(category).toArray()
    return list.filter(p => p.isPublished && !p.deleted)
  },
  bulkSave: async (products) => {
    if (!products.length) return
    await db.transaction('rw', db.products, async () => {
      for (const p of products) {
        await db.products.put({ ...p, synced: 1 })
      }
    })
  }
}

const loadingCheckout = ref(false)

const finalTotal = computed(() => cartStore.totalPrice + deliveryFee.value)

onMounted(async () => {
  await cartStore.loadCart()
  const remote = await qrisService.fetchRemoteSettings()
  if (remote) qrisSettings.value = remote
})

async function handleFileChange(event) {
  const file = event.target.files[0]
  if (!file) return
  uploading.value = true
  try {
    const compressedBlob = await compressImage(file)
    // INI KUNCINYA MET -> OTOMATIS MASUK FOLDER sinar-pagi/bukti-bayar
    const res = await uploadBuktiBayar(compressedBlob)
    buktiUrl.value = res.secure_url
    buktiFile.value = file.name
  } catch (err) {
    console.error('Gagal upload bukti bayar:', err)
    alert('Gagal mengunggah bukti pembayaran. Silakan coba lagi.')
  } finally {
    uploading.value = false
  }
}

async function downloadQris() {
  if (!qrisSettings.value.qrisUrl) return
  try {
    const res = await fetch(qrisSettings.value.qrisUrl)
    const blob = await res.blob()
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `QRIS-${qrisSettings.value.merchantName || 'SinarPagi'}.jpg`
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    URL.revokeObjectURL(url)
  } catch {
    window.open(qrisSettings.value.qrisUrl, '_blank')
  }
}

async function handleCheckout() {
  if (!customerName.value.trim()) {
    alert('Mohon isi Nama terlebih dahulu!')
    return
  }
  if (deliveryType.value === 'delivery' && !customerAddress.value.trim()) {
    alert('Mohon isi Alamat lengkap untuk pengantaran!')
    return
  }
  if (deliveryType.value === 'pickup' && !customerAddress.value.trim()) {
    customerAddress.value = 'Ambil di Warung'
  }
  if (paymentMethod.value === 'QRIS' && !buktiUrl.value) {
    alert('Mohon upload bukti pembayaran QRIS terlebih dahulu!')
    return
  }

  loadingCheckout.value = true

  waService.sendOrderToWhatsApp(
    { 
      name: customerName.value, 
      address: customerAddress.value,
      deliveryType: deliveryType.value,
      deliveryFee: deliveryFee.value
    },
    cartStore.items,
    finalTotal.value,
    paymentMethod.value,
    buktiUrl.value
  )

  setTimeout(() => {
    loadingCheckout.value = false
    router.push('/checkout-success')
  }, 1000)
}

async function openProductOptions(product) {
  let pkgs = []
  let services = []

  try {
  const allPkgs = await db.product_packages.where('productId').equals(product.id).toArray()
  pkgs = allPkgs.filter(p => {
    const name = (p.name || '').toUpperCase()
    const isBelanjaBesar = ['RENCENG', 'RENCENGAN', 'DUS', 'KARDUS'].some(bad => name.includes(bad))
    const isSatuBatang = (p.qty_pcs || 0) === 1 // HIDE DI WEB AJA
    return !isBelanjaBesar && !isSatuBatang
  })
} catch { pkgs = [] }

  try {
    const allServices = await db.services.toArray()
    // FIX ANTI BOCOR
    const getIds = (s) => {
      if (s.productIds && Array.isArray(s.productIds) && s.productIds.length) return s.productIds
      if (s.productId) return [s.productId]
      return []
    }
    
    const custom = allServices.filter(s => getIds(s).includes(product.id))
    if (custom.length > 0) {
      services = custom
    } else {
      const cat = (product.category || '').toUpperCase()
      services = allServices.filter(s => {
        const ids = getIds(s)
        return ids.length === 0 && (s.category || '').toUpperCase() === cat
      })
    }
  } catch { services = [] }

  const hasOptions = pkgs.length > 0 || services.length > 0

  if (!hasOptions) {
    const itemToAdd = buildCartItem(product, [], product.price_sell, null)
    cartStore.addToCart(itemToAdd, 1)
    return
  }

  activeProduct.value = product
  selectedPackage.value = null
  selectedServices.value = []
  productPackages.value = pkgs
  availableServices.value = services
  showModal.value = true
}

const totalPrice = computed(() => {
  if (!activeProduct.value) return 0
  let price = activeProduct.value.price_sell
  if (selectedPackage.value) {
    price = selectedPackage.value.price_sell
  }
  for (const svc of selectedServices.value) {
    price += Number(svc.price) || 0
  }
  return price
})

function toggleService(svc) {
  const idx = selectedServices.value.findIndex(s => s.id === svc.id)
  if (idx >= 0) selectedServices.value.splice(idx, 1)
  else selectedServices.value.push(svc)
}

function addToCartWithOptions() {
  if (!activeProduct.value) return
  const itemToAdd = buildCartItem(activeProduct.value, selectedServices.value, totalPrice.value, selectedPackage.value)
  cartStore.addToCart(itemToAdd, 1)
  showModal.value = false
  activeProduct.value = null
}
