import { ref } from 'vue'

export function useCartAnimation() {
  const isCartBouncing = ref(false)
  const addedProductId = ref(null)

  /**
   * Menjalankan animasi micro-feedback di tombol + bounce pada keranjang
   * @param {String|Number} productId - ID produk yang baru ditambahkan
   * @param {Number} duration - Durasi animasi dalam ms
   */
  function triggerCartAnimation(productId = null, duration = 600) {
    // 1. Set ID produk untuk menampilkan badge / micro-toast "✓ Ditambahkan"
    if (productId) {
      addedProductId.value = productId
    }

    // 2. Trigger efek bounce pada keranjang
    isCartBouncing.value = true

    // 3. Reset state setelah durasi selesai
    setTimeout(() => {
      isCartBouncing.value = false
      addedProductId.value = null
    }, duration)
  }

  return {
    isCartBouncing,
    addedProductId,
    triggerCartAnimation
  }
}

import { db } from '../index.js'

export const productRepo = {
  getPublished: async () => {
    const list = await db.products.where('isPublished').equals(1).toArray()
    return list.filter(p => !p.deleted)
  },
  getById: async (id) => {
    const item = await db.products.get(id)
    return item && item.isPublished && !item.deleted ? item : null
  },
  getByCategory: async (category) => {
    if (!category || category === 'SEMUA') return productRepo.getPublished()
    const list = await db.products.where('category').equals(category).toArray()
    return list.filter(p => p.isPublished && !p.deleted)
  },
  bulkSave: async (products) => {
    if (!products.length) return
    await db.transaction('rw', db.products, async () => {
      for (const p of products) {
        await db.products.put({ ...p, synced: 1 })
      }
    })
  },
  // MEWADAHI QUERY PACKAGES AGAR VIEW TIDAK TOUCH DB
  getPackagesByProductId: async (productId) => {
    const allPkgs = await db.product_packages.where('productId').equals(productId).toArray()
    return allPkgs.filter(p => !p.deleted)
  },
  // MEWADAHI QUERY SERVICES AGAR VIEW TIDAK TOUCH DB
  getAllServices: async () => {
    const allServices = await db.services.toArray()
    return allServices.filter(s => !s.deleted)
  }
}

/**
 * Upload ke Cloudinary - folder auto kebuat pas upload pertama
 * @param {Blob} imageBlob
 * @param {String} folder - 'sinar-pagi/products' | 'sinar-pagi/qris' | 'sinar-pagi/bukti-bayar'
 */
export async function uploadToCloudinary(imageBlob, folder = 'sinar-pagi/bukti-bayar') {
  const formData = new FormData()
  formData.append('file', imageBlob)
  formData.append('upload_preset', UPLOAD_PRESET)
  formData.append('folder', folder) // INI YANG BIKIN AUTO FOLDER

  const response = await fetch(`https://api.cloudinary.com/v1_1/${CLOUD_NAME}/image/upload`, {
    method: 'POST',
    body: formData
  })
  if (!response.ok) {
    const errorText = await response.text()
    throw new Error(`Cloudinary Upload Failed: ${errorText}`)
  }
  const data = await response.json()
  return {
    secure_url: data.secure_url,
    public_id: data.public_id
  }
}

import { productRepo } from '../db/repositories/productRepository.js'
import { db } from '../db/index.js'
import { supabase } from './supabase.js'

const getServiceIds = (s) => {
  if (s.productIds && Array.isArray(s.productIds) && s.productIds.length) return s.productIds
  if (s.productId) return [s.productId]
  if (s.product_id) return [s.product_id]
  return []
}

const VERSION_KEY = 'sinar_pagi_catalog_version'
const LAST_PULL_KEY = 'sinar_pagi_last_pull'

export const productService = {
  async syncPublishedProducts() {
    try {
      if (!navigator.onLine) {
        return await productRepo.getPublished()
      }

      // 1. CEK VERSI (Melihat updated_at paling baru dari Supabase)
      let remoteVersion = null
      try {
        const { data, error } = await supabase
          .from('products')
          .select('updated_at')
          .order('updated_at', { ascending: false })
          .limit(1)

        if (!error && data && data.length) {
          remoteVersion = data[0].updated_at
        }
      } catch (e) {
        console.warn('[ProductService] Gagal cek meta version', e)
      }

      // 2. KALO VERSI SAMA -> CACHE HIT (0 Rows Read)
      if (remoteVersion) {
        try {
          const localVersion = localStorage.getItem(VERSION_KEY)
          if (localVersion && localVersion === String(remoteVersion)) {
            const localData = await productRepo.getPublished()
            if (localData.length > 0) {
              console.log('[ProductService] CACHE HIT - Memuat data dari Dexie lokal!')
              return localData
            }
          }
        } catch {}
      }

      const lastPull = localStorage.getItem(LAST_PULL_KEY)

      // 3. DELTA SYNC (Cuma ambil data yang berubah sejak lastPull)
      if (lastPull && remoteVersion) {
        console.log(`[ProductService] DELTA SYNC since ${lastPull}...`)
        let totalReads = 0

        // Sync Delta Products
        try {
          const { data: prodData, error } = await supabase
            .from('products')
            .select('*')
            .gt('updated_at', lastPull)

          if (error) throw error
          totalReads += prodData?.length || 0

          if (prodData && prodData.length) {
            const changed = prodData.map(d => ({
              id: d.id,
              name: d.name || '',
              category: d.category || 'UMUM',
              unit: d.unit || 'PCS',
              price_sell: Number(d.price_sell || d.price || 0),
              price_modal: Number(d.price_modal || 0),
              packPrice: Number(d.pack_price || d.packPrice || 0),
              packQty: Number(d.pack_qty || d.packQty || 0),
              purchasePackName: d.purchase_pack_name || d.purchasePackName || '',
              qty: Number(d.qty || d.stock || 0),
              minStock: Number(d.min_stock || d.minStock || 2),
              isPublished: (d.is_published ?? d.isPublished ?? true) ? 1 : 0,
              imageUrl: d.image_url || d.imageUrl || null,
              imagePublicId: d.image_public_id || d.imagePublicId || null,
              updatedAt: d.updated_at || d.updatedAt || new Date().toISOString(),
              updated_at: d.updated_at || d.updatedAt || new Date().toISOString(),
              deleted: d.deleted ? 1 : 0,
              synced: 1
            }))
            await db.products.bulkPut(changed)
          }
        } catch (e) {
          console.warn('[ProductService] Delta products gagal', e)
        }

        // Sync Delta Packages
        try {
          const { data: pkgData, error } = await supabase
            .from('product_packages')
            .select('*')
            .gt('updated_at', lastPull)

          if (error) throw error
          totalReads += pkgData?.length || 0

          if (pkgData && pkgData.length) {
            const changedPkg = pkgData.map(d => ({
              id: d.id,
              productId: d.product_id || d.productId,
              name: d.name,
              qty_pcs: d.qty_pcs || d.multiplier || 1,
              price_sell: Number(d.price_sell) || 0,
              order: d.order || 0,
              updatedAt: d.updated_at || d.updatedAt || new Date().toISOString(),
              updated_at: d.updated_at || d.updatedAt || new Date().toISOString(),
              deleted: d.deleted ? 1 : 0,
              synced: 1
            }))
            const toPut = changedPkg.filter(p => !p.deleted)
            if (toPut.length) await db.product_packages.bulkPut(toPut)
          }
        } catch (e) {
          console.warn('[ProductService] Delta packages gagal', e)
        }

        // Simpan versi + timestamp
        try {
          if (remoteVersion) localStorage.setItem(VERSION_KEY, String(remoteVersion))
          localStorage.setItem(LAST_PULL_KEY, new Date().toISOString())
        } catch {}

        return await productRepo.getPublished()
      }

      // 4. FULL SYNC PERTAMA KALI
      console.log('[ProductService] Full Sync Produk dari Supabase...')

      const { data: snapshot, error } = await supabase
        .from('products')
        .select('*')
        .eq('is_published', true)

      if (!error && snapshot) {
        const remoteProducts = snapshot
          .filter(d => !d.deleted)
          .map(d => ({
            id: d.id,
            name: d.name || '',
            category: d.category || 'UMUM',
            unit: d.unit || 'PCS',
            price_sell: Number(d.price_sell || d.price || 0),
            price_modal: Number(d.price_modal || 0),
            packPrice: Number(d.pack_price || d.packPrice || 0),
            packQty: Number(d.pack_qty || d.packQty || 0),
            purchasePackName: d.purchase_pack_name || d.purchasePackName || '',
            qty: Number(d.qty || d.stock || 0),
            minStock: Number(d.min_stock || d.minStock || 2),
            isPublished: (d.is_published ?? d.isPublished ?? true) ? 1 : 0,
            imageUrl: d.image_url || d.imageUrl || null,
            imagePublicId: d.image_public_id || d.imagePublicId || null,
            updatedAt: d.updated_at || d.updatedAt || new Date().toISOString(),
            updated_at: d.updated_at || d.updatedAt || new Date().toISOString(),
            deleted: 0,
            synced: 1
          }))

        if (remoteProducts.length > 0) await productRepo.bulkSave(remoteProducts)
      }

      // Full Sync Packages & Services
      try {
        const { data: pkgSnap } = await supabase.from('product_packages').select('*')
        if (pkgSnap?.length) {
          const remotePkgs = pkgSnap
            .filter(p => !p.deleted)
            .map(d => ({
              id: d.id,
              productId: d.product_id || d.productId,
              name: d.name,
              qty_pcs: d.qty_pcs || d.multiplier || 1,
              price_sell: Number(d.price_sell) || 0,
              order: d.order || 0,
              updatedAt: d.updated_at || d.updatedAt || new Date().toISOString(),
              updated_at: d.updated_at || d.updatedAt || new Date().toISOString(),
              deleted: 0,
              synced: 1
            }))
          await db.product_packages.bulkPut(remotePkgs)
        }
      } catch (e) {
        console.warn('[ProductService] Full sync packages gagal', e)
      }

      try {
        const { data: svcSnap } = await supabase.from('services').select('*')
        if (svcSnap?.length) {
          const remoteSvcs = svcSnap
            .filter(d => !d.deleted)
            .map(d => ({
              id: d.id,
              name: d.name || '',
              price: Number(d.price) || 0,
              productId: d.productId || d.product_id || null,
              productIds: getServiceIds(d),
              category: (d.category || 'UMUM').toUpperCase(),
              updatedAt: d.updated_at || d.updatedAt || new Date().toISOString(),
              updated_at: d.updated_at || d.updatedAt || new Date().toISOString(),
              deleted: 0,
              synced: 1
            }))
          await db.services.bulkPut(remoteSvcs)
        }
      } catch (e) {
        console.warn('[ProductService] Full sync services gagal', e)
      }

      // Simpan LocalStorage Cache Meta
      try {
        if (remoteVersion) {
          localStorage.setItem(VERSION_KEY, String(remoteVersion))
        } else {
          localStorage.setItem(VERSION_KEY, String(Date.now()))
        }
        localStorage.setItem(LAST_PULL_KEY, new Date().toISOString())
      } catch {}

      return await productRepo.getPublished()
    } catch (error) {
      console.error('[ProductService] Error:', error)
      return await productRepo.getPublished()
    }
  }
}

import { supabase } from './supabase.js'

const QRIS_CACHE_KEY = 'sinar_pagi_qris_cache'
const QRIS_CACHE_TIME_KEY = 'sinar_pagi_qris_cache_time'
const QRIS_CACHE_DURATION = 60 * 60 * 1000 // 1 jam cache

export const qrisService = {
  async fetchRemoteSettings() {
    if (!navigator.onLine) {
      try {
        const cached = localStorage.getItem(QRIS_CACHE_KEY)
        if (cached) return JSON.parse(cached)
      } catch {}
      return null
    }

    // Hemat request API: Cek cache dulu
    try {
      const lastCache = Number(localStorage.getItem(QRIS_CACHE_TIME_KEY) || 0)
      const now = Date.now()
      if (lastCache && (now - lastCache < QRIS_CACHE_DURATION)) {
        const cached = localStorage.getItem(QRIS_CACHE_KEY)
        if (cached) {
          return JSON.parse(cached)
        }
      }
    } catch {}

    try {
      // 1. Coba cari setting khusus id = 'qris'
      let { data, error } = await supabase
        .from('settings')
        .select('*')
        .eq('id', 'qris')
        .maybeSingle()

      // 2. Fallback jika tersimpan di id = 'default_settings'
      if (!data) {
        const fallbackRes = await supabase
          .from('settings')
          .select('*')
          .limit(1)
          .maybeSingle()
        data = fallbackRes.data
      }

      if (data) {
        const result = {
          id: data.id || 'qris',
          qrisUrl: data.qris_url || data.qrisUrl || data.image_url || '',
          merchantName: data.merchant_name || data.merchantName || 'WARUNG SINAR PAGI',
          nmid: data.nmid || '',
          updatedAt: data.updated_at || new Date().toISOString()
        }

        try {
          localStorage.setItem(QRIS_CACHE_KEY, JSON.stringify(result))
          localStorage.setItem(QRIS_CACHE_TIME_KEY, String(Date.now()))
        } catch {}

        return result
      }
    } catch (error) {
      console.error('[QrisService] Error fetch QRIS:', error)
      try {
        const cached = localStorage.getItem(QRIS_CACHE_KEY)
        if (cached) return JSON.parse(cached)
      } catch {}
    }
    return null
  }
}

export const useCartStore = defineStore('cart', () => {

    items.value = await cartRepo.getAll()
  }

  async function addToCart(product, qty = 1) {
    const plainProduct = JSON.parse(JSON.stringify(product))
    
    const existing = items.value.find(i => 
      i.productId === plainProduct.id && 
      i.packageId === (plainProduct.packageId || null) &&
      JSON.stringify(i.services || []) === JSON.stringify(plainProduct.services || [])
    )
    
    if (existing) {
      existing.qty += qty
      existing.updatedAt = new Date().toISOString()
      await cartRepo.save(existing)
    } else {
      const newItem = {
        id: crypto.randomUUID(),
        productId: plainProduct.id,
        name: plainProduct.name,
        originalName: plainProduct.originalName || plainProduct.name,
        price_sell: Number(plainProduct.price_sell) || 0,
        imageUrl: plainProduct.imageUrl || null,
        qty: qty,
        unit: plainProduct.unit || 'PCS',
        note: plainProduct.note || '',
        packageId: plainProduct.packageId || null,
        services: plainProduct.services || [],
        category: plainProduct.category || 'UMUM',
        updatedAt: new Date().toISOString()
      }
      await cartRepo.save(newItem)
    }
    await loadCart()
  }

  // FIX: pake cartId (id random), bukan productId
  async function updateQty(cartId, qty) {
    if (qty <= 0) {
      await cartRepo.removeById(cartId)
    } else {
      await cartRepo.updateQtyById(cartId, qty)
    }
    await loadCart()
  }

  async function removeFromCart(cartId) {
    await cartRepo.removeById(cartId)
    await loadCart()
  }

  async function clearCart() {
    await cartRepo.clear()
    items.value = []
  }

  const totalItems = computed(() => items.value.reduce((sum, i) => sum + i.qty, 0))
  const totalPrice = computed(() => items.value.reduce((sum, i) => sum + (i.price_sell * i.qty), 0))

  return { items, loadCart, addToCart, updateQty, removeFromCart, clearCart, totalItems, totalPrice }
})

import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { productService } from '../services/productService'
import { productRepo } from '../db/repositories/productRepository'
import { PINNED_CATEGORIES, VISIBLE_COUNT } from '../config/catalogCategory.js'

export const useProductStore = defineStore('product', () => {
  const products = ref([])
  const loading = ref(false)
  const searchQuery = ref('')
  const selectedCategory = ref('SEMUA')

  // AMBIL PINNED DARI LOCALSTORAGE DULU, KALO GA ADA BARU DARI CONFIG FILE
  function getPinnedCategories() {
    try {
      const fromLS = JSON.parse(localStorage.getItem('sinar_pagi_pinned_categories') || 'null')
      if (fromLS && Array.isArray(fromLS) && fromLS.length > 0) return fromLS
    } catch {}
    return PINNED_CATEGORIES
  }

  const categories = computed(() => {
    const cats = new Set(products.value.map(p => p.category || 'UMUM'))
    return ['SEMUA', ...Array.from(cats)]
  })

  // MEMINDAHKAN LOGIKA POPULAR CATEGORIES DARI CATALOG.VUE KE STORE
  const popularCategories = computed(() => {
    const all = categories.value || []
    const catsOnly = all.filter(c => c !== 'SEMUA' && c.toUpperCase() !== 'SEMUA')
    const pinned = getPinnedCategories()

    // KALO PINNED ADA ISINYA, PAKE PINNED SESUAI URUTAN
    if (pinned && pinned.length > 0) {
      const valid = pinned.filter(p => catsOnly.includes(p))
      if (valid.length > 0) return valid.slice(0, VISIBLE_COUNT)
    }

    // KALO PINNED KOSONG, FALLBACK AUTO PALING LARIS (JUMLAH PRODUK TERBANYAK)
    const counts = {}
    if (products.value.length === 0) {
      return catsOnly.sort((a, b) => a.localeCompare(b)).slice(0, VISIBLE_COUNT)
    }
    products.value.forEach(p => {
      const cat = p.category || 'UMUM'
      counts[cat] = (counts[cat] || 0) + 1
    })
    return catsOnly.sort((a, b) => (counts[b] || 0) - (counts[a] || 0)).slice(0, VISIBLE_COUNT)
  })

  const visibleCategories = computed(() => {
    return ['SEMUA', ...popularCategories.value]
  })

  const remainingCategories = computed(() => {
    const visibleSet = new Set(visibleCategories.value)
    return categories.value.filter(c => !visibleSet.has(c))
  })

  const filteredProducts = computed(() => {
    return products.value.filter(p => {
      const matchSearch = p.name.toLowerCase().includes(searchQuery.value.toLowerCase())
      const matchCategory = selectedCategory.value === 'SEMUA' || p.category === selectedCategory.value
      return matchSearch && matchCategory
    })
  })

  async function loadProducts() {
    loading.value = true
    try {
      const data = await productService.syncPublishedProducts()
      products.value = data || []
    } catch (err) {
      console.error('[ProductStore] Gagal memuat produk:', err)
      products.value = await productRepo.getPublished()
    } finally {
      loading.value = false
    }
  }

  // HELPER UNTUK OPSI PRODUK LEWAT REPOSITORY (VIEW TANPA TOUCH DB)
  async function fetchProductPackages(productId) {
    return await productRepo.getPackagesByProductId(productId)
  }

  async function fetchAllServices() {
    return await productRepo.getAllServices()
  }

  return {
    products,
    loading,
    searchQuery,
    selectedCategory,
    categories,
    popularCategories,
    visibleCategories,
    remainingCategories,
    filteredProducts,
    loadProducts,
    fetchProductPackages,
    fetchAllServices
  }
})

/**
 * Utility Kompresi Gambar via Canvas (800px JPEG 70%)
 * @param {File} file 
 * @returns {Promise<Blob>}
 */
export function compressImage(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = (event) => {
      const img = new Image()
      img.onload = () => {
        const canvas = document.createElement('canvas')
        const maxWidth = 800
        let width = img.width
        let height = img.height
        if (width > maxWidth) {
          height = Math.round((height * maxWidth) / width)
          width = maxWidth
        }
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        ctx.drawImage(img, 0, 0, width, height)
        canvas.toBlob(
          (blob) => (blob ? resolve(blob) : reject(new Error('Gagal kompres'))),
          'image/jpeg',
          0.7
        )
      }
      img.onerror = (err) => reject(err)
      img.src = event.target.result
    }
    reader.onerror = (err) => reject(err)
    reader.readAsDataURL(file)
  })
}

export function buildCartItem(product, selectedServices = [], finalPrice = null, selectedPackage = null) {
  let finalUnit = product.unit || 'PCS'
  let displayName = product.name
  let notes = []

  if (selectedPackage) {
    if (selectedPackage.id) {
      finalUnit = selectedPackage.name || product.purchasePackName || 'PAKET'
      // FIX MET - CUMAN (1 BKS) DOANG, GA USAH 12 BATANG LAGI
      displayName = `${product.name} (${selectedPackage.name})`
    } else if (typeof selectedPackage === 'object' && selectedPackage.name) {
      finalUnit = selectedPackage.name
      displayName = `${product.name} (${selectedPackage.name})`
    }
  }

  if (Array.isArray(selectedServices) && selectedServices.length > 0) {
    notes = selectedServices.map(s => s.name)
  }

  const priceToUse = finalPrice != null ? finalPrice : product.price_sell

  return {
    id: product.id,
    name: displayName,
    originalName: product.name,
    price_sell: Number(priceToUse) || 0,
    imageUrl: product.imageUrl,
    category: product.category,
    unit: finalUnit,
    note: notes.join(', '),
    // CLONE PLAIN BIAR GA DataCloneError DI INDEXEDDB
    services: JSON.parse(JSON.stringify(selectedServices || [])),
    packageId: selectedPackage?.id || null,
    qty: product.qty
  }
}

export function getServicesForProduct(product, allServices = []) {
  if (!product) return []
  const list = allServices.filter(s => !s.deleted)
  const pCat = (product.category || '').toUpperCase()

  // 1. Cari yang khusus buat produk ini (support multi)
  const custom = list.filter(s => {
    const ids = getServiceIds(s)
    return ids.length > 0 && ids.includes(product.id)
  })
  // Kalo ada khusus, CUMA khusus yang keluar, umum HILANG
  if (custom.length > 0) return custom

  // 2. Kalo ga ada khusus, baru umum yang ids nya kosong beneran
  return list.filter(s => {
    const ids = getServiceIds(s)
    return ids.length === 0 && (s.category || '').toUpperCase() === pCat
  })
}

export function slugify(text) {
  return text
    .toString()
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9\s-]/g, '')
    .replace(/\s+/g, '-')
    .replace(/-+/g, '-')
}

export function makeProductSlug(product) {
  return `${slugify(product.name)}-${product.id}`
}

export function parseProductIdFromSlug(slug) {
  if (!slug) return null
  const parts = slug.split('-')
  return parts[parts.length - 1]
}

<script setup>
import { ref, onMounted, computed } from 'vue'
import { useRouter } from 'vue-router'
import { useCartStore } from '../stores/cartStore'
import { qrisService } from '../services/qrisService'
import { waService } from '../services/waService'
import { compressImage, uploadBuktiBayar } from '../services/cloudinaryService'
import { formatRupiah } from '../utils/formatters'
import { ArrowLeft, User, MapPin, QrCode, Banknote, ShieldCheck, Upload, CheckCircle, Loader2, Store, Bike, Download, Maximize2, X, StickyNote } from 'lucide-vue-next'

const router = useRouter()
const cartStore = useCartStore()

const customerName = ref('')
const customerAddress = ref('') // Alamat / Patokan WAJIB
const customerNotes = ref('') // Catatan / Notes OPSIONAL
const paymentMethod = ref('COD')
const deliveryType = ref('pickup') // pickup | delivery

const isQrisFullscreen = ref(false)
const qrisSettings = ref({ qrisUrl: '', merchantName: 'WARUNG SINAR PAGI', nmid: '' })
const buktiFile = ref(null)
const buktiUrl = ref('')
const uploading = ref(false)
const loadingCheckout = ref(false)

const DELIVERY_FEE = 2000

const deliveryFee = computed(() => deliveryType.value === 'delivery' ? DELIVERY_FEE : 0)
const finalTotal = computed(() => cartStore.totalPrice + deliveryFee.value)

onMounted(async () => {
  await cartStore.loadCart()
  const remote = await qrisService.fetchRemoteSettings()
  if (remote) qrisSettings.value = remote
})

async function handleFileChange(event) {
  const file = event.target.files[0]
  if (!file) return
  uploading.value = true
  try {
    const compressedBlob = await compressImage(file)
    // INI KUNCINYA MET -> OTOMATIS MASUK FOLDER sinar-pagi/bukti-bayar
    const res = await uploadBuktiBayar(compressedBlob)
    buktiUrl.value = res.secure_url
    buktiFile.value = file.name
  } catch (err) {
    console.error('Gagal upload bukti bayar:', err)
    alert('Gagal mengunggah bukti pembayaran. Silakan coba lagi.')
  } finally {
    uploading.value = false
  }
}

async function downloadQris() {
  if (!qrisSettings.value.qrisUrl) return
  try {
    const res = await fetch(qrisSettings.value.qrisUrl)
    const blob = await res.blob()
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `QRIS-${qrisSettings.value.merchantName || 'SinarPagi'}.jpg`
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    URL.revokeObjectURL(url)
  } catch {
    window.open(qrisSettings.value.qrisUrl, '_blank')
  }
}

async function handleCheckout() {
  if (!customerName.value.trim()) {
    alert('Mohon isi Nama Lengkap terlebih dahulu!')
    return
  }
  if (!customerAddress.value.trim()) {
    alert('Mohon isi Alamat / Patokan terlebih dahulu!')
    return
  }
  if (paymentMethod.value === 'QRIS' && !buktiUrl.value) {
    alert('Mohon upload bukti pembayaran QRIS terlebih dahulu!')
    return
  }

  loadingCheckout.value = true

  waService.sendOrderToWhatsApp(
    { 
      name: customerName.value, 
      address: customerAddress.value,
      notes: customerNotes.value,
      deliveryType: deliveryType.value,
      deliveryFee: deliveryFee.value
    },
    cartStore.items,
    finalTotal.value,
    paymentMethod.value,
    buktiUrl.value
  )

  setTimeout(() => {
    loadingCheckout.value = false
    router.push('/checkout-success')
  }, 1000)
}
</script>

<script setup>
import { ref, onMounted, computed } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { useProductStore } from '../stores/productStore'
import { useCartStore } from '../stores/cartStore'
import { getThumbUrl } from '../services/cloudinaryService'
import { formatRupiah } from '../utils/formatters'
import { buildCartItem } from '../utils/productHelper'
import { parseProductIdFromSlug, makeProductSlug } from '../utils/slugify'
import { useHead } from '@vueuse/head'
import { ArrowLeft, Package, ShoppingCart } from 'lucide-vue-next'

const route = useRoute()
const router = useRouter()
const productStore = useProductStore()
const cartStore = useCartStore()

const product = ref(null)
const loading = ref(true)
const notFound = ref(false)

const productPackages = ref([])
const availableServices = ref([])
const selectedPackage = ref(null)
const selectedServices = ref([])

const totalPrice = computed(() => {
  if (!product.value) return 0
  let price = product.value.price_sell
  if (selectedPackage.value) price = selectedPackage.value.price_sell
  for (const svc of selectedServices.value) price += Number(svc.price) || 0
  return price
})

function toggleService(svc) {
  const exists = selectedServices.value.find(s => s.id === svc.id)
  if (exists) selectedServices.value = []
  else selectedServices.value = [svc]
}

onMounted(async () => {
  loading.value = true
  await productStore.loadProducts()
  
  const slugParam = route.params.slug || ''
  
  // TAHAP 6 FIX: CARI DENGAN 3 CARA BIAR GAK ADA YANG "TIDAK DITEMUKAN" LAGI
  let found = null
  
  // Cara 1: Cocok persis slug lengkap
  found = productStore.products.find(p => makeProductSlug(p) === slugParam)
  
  // Cara 2: Cari yang ID nya jadi akhiran slug (PALING AMAN, anti kepotong -)
  if (!found) {
    found = productStore.products.find(p => slugParam.endsWith(p.id))
  }
  
  // Cara 3: Fallback cara lama
  if (!found) {
    const id = parseProductIdFromSlug(slugParam)
    found = productStore.products.find(p => p.id === id)
  }
  
  if (!found) {
    notFound.value = true
    loading.value = false
    return
  }
  
  product.value = found

  // ambil paket & layanan dari IndexedDB (0 read)
  try {
    const pkgs = await productStore.fetchProductPackages(found.id)
    productPackages.value = pkgs.filter(p => {
      const name = (p.name || '').toUpperCase()
      return !['RENCENG','RENCENGAN','DUS','KARDUS'].some(bad => name.includes(bad)) && (p.qty_pcs || 0) !== 1
    }).sort((a,b) => (a.order ?? 999) - (b.order ?? 999))
  } catch {}
  try {
    const allServices = await productStore.fetchAllServices()
    const getIds = (s) => s.productIds?.length ? s.productIds : s.productId ? [s.productId] : []
    const custom = allServices.filter(s => getIds(s).includes(found.id))
    if (custom.length) availableServices.value = custom
    else {
      const cat = (found.category || '').toUpperCase()
      availableServices.value = allServices.filter(s => getIds(s).length === 0 && (s.category || '').toUpperCase() === cat)
    }
  } catch {}

  // SEO DYNAMIC
  const title = `${found.name} - Warung Sinar Pagi Ciheleut Bogor | Warkop & Sembako Diantar 2rb`
  const desc = `Beli ${found.name} di Warung Sinar Pagi Jl. Jagung Ujung No.38 Ciheleut Bogor Timur. ${found.category || ''} - Stok ready, buka 09.00-21.00, diantar 200m cuma Rp2.000 COD/QRIS. Harga ${formatRupiah(found.price_sell)}`
  const url = `https://warungsinarpagi.my.id/produk/${makeProductSlug(found)}`
  const image = found.imageUrl ? getThumbUrl(found.imageUrl, 600) : ''

  useHead({
    title,
    meta: [
      { name: 'description', content: desc },
      { property: 'og:title', content: title },
      { property: 'og:description', content: desc },
      { property: 'og:image', content: image },
      { property: 'og:url', content: url },
      { property: 'og:type', content: 'product' }
    ],
    script: [
      {
        type: 'application/ld+json',
        innerHTML: JSON.stringify({
          "@context": "https://schema.org",
          "@type": "Product",
          "name": found.name,
          "image": found.imageUrl ? [found.imageUrl] : [],
          "description": desc,
          "sku": found.id,
          "brand": { "@type": "Brand", "name": "Warung Sinar Pagi" },
          "category": found.category,
          "offers": {
            "@type": "Offer",
            "url": url,
            "priceCurrency": "IDR",
            "price": found.price_sell,
            "priceValidUntil": new Date(Date.now() + 30*24*60*60*1000).toISOString().split('T')[0],
            "availability": found.qty > 0 ? "https://schema.org/InStock" : "https://schema.org/OutOfStock",
            "seller": {
              "@type": "Organization",
              "name": "Warung Sinar Pagi",
              "address": {
                "@type": "PostalAddress",
                "streetAddress": "Jl. Jagung Ujung No.38 RT01/09",
                "addressLocality": "Ciheleut",
                "addressRegion": "Bogor",
                "postalCode": "16142"
              }
            }
          }
        })
      },
      {
        type: 'application/ld+json',
        innerHTML: JSON.stringify({
          "@context": "https://schema.org",
          "@type": "BreadcrumbList",
          "itemListElement": [
            { "@type": "ListItem", "position": 1, "name": "Beranda", "item": "https://warungsinarpagi.my.id/" },
            { "@type": "ListItem", "position": 2, "name": found.category || "Produk", "item": `https://warungsinarpagi.my.id/#${found.category}` },
            { "@type": "ListItem", "position": 3, "name": found.name }
          ]
        })
      }
    ]
  })

  loading.value = false
})

function addToCart() {
  if (!product.value) return
  const item = buildCartItem(product.value, selectedServices.value, totalPrice.value, selectedPackage.value)
  cartStore.addToCart(item, 1)
  router.push('/cart')
}
