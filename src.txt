import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { productService } from '../services/productService'
import { productRepo } from '../db/repositories/productRepository'

export const useProductStore = defineStore('product', () => {
  const products = ref([])
  const loading = ref(false)
  const searchQuery = ref('')
  const selectedCategory = ref('SEMUA')

  const categories = computed(() => {
    const cats = new Set(products.value.map(p => p.category || 'UMUM'))
    return ['SEMUA', ...Array.from(cats)]
  })

  const filteredProducts = computed(() => {
    return products.value.filter(p => {
      const matchSearch = p.name.toLowerCase().includes(searchQuery.value.toLowerCase())
      const matchCategory = selectedCategory.value === 'SEMUA' || p.category === selectedCategory.value
      return matchSearch && matchCategory
    })
  })

  async function loadProducts() {
    loading.value = true
    try {
      const data = await productService.syncPublishedProducts()
      products.value = data || []
    } catch (err) {
      console.error('[ProductStore] Gagal memuat produk:', err)
      products.value = await productRepo.getPublished()
    } finally {
      loading.value = false
    }
  }

  return {
    products,
    loading,
    searchQuery,
    selectedCategory,
    categories,
    filteredProducts,
    loadProducts
  }
})

===== src/stores/cartStore.js =====
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'
import { cartRepo } from '../db/repositories/cartRepository'

export const useCartStore = defineStore('cart', () => {
  const items = ref([])

  async function loadCart() {
    items.value = await cartRepo.getAll()
  }

  async function addToCart(product, qty = 1) {
    const existing = items.value.find(i => i.productId === product.id)
    if (existing) {
      existing.qty += qty
      await cartRepo.save(existing)
    } else {
      const newItem = {
        id: crypto.randomUUID(),
        productId: product.id,
        name: product.name,
        price_sell: product.price_sell,
        imageUrl: product.imageUrl,
        qty: qty,
        unit: product.unit || 'PCS',
        note: product.note || '',
        updatedAt: new Date().toISOString()
      }
      await cartRepo.save(newItem)
    }
    await loadCart()
  }

  async function updateQty(productId, qty) {
    await cartRepo.updateQty(productId, qty)
    await loadCart()
  }

  async function removeFromCart(productId) {
    await cartRepo.remove(productId)
    await loadCart()
  }

  async function clearCart() {
    await cartRepo.clear()
    items.value = []
  }

  const totalItems = computed(() => items.value.reduce((sum, i) => sum + i.qty, 0))
  const totalPrice = computed(() => items.value.reduce((sum, i) => sum + (i.price_sell * i.qty), 0))

  return {
    items,
    loadCart,
    addToCart,
    updateQty,
    removeFromCart,
    clearCart,
    totalItems,
    totalPrice
  }
})

===== src/services/productService.js =====
import { collection, getDocs, query, where } from 'firebase/firestore'
import { db as firestoreDb } from './firebase.js'
import { productRepo } from '../db/repositories/productRepository.js'

export const productService = {
  async syncPublishedProducts() {
    try {
      if (!navigator.onLine) {
        console.log('[ProductService] Offline, pake IndexedDB')
        return await productRepo.getPublished()
      }
      const q = query(collection(firestoreDb, 'products'), where('isPublished', '==', true))
      const snapshot = await getDocs(q)
      const remoteProducts = []
      snapshot.forEach(doc => {
        const data = doc.data()
        if (data.deleted) return
        remoteProducts.push({
          id: doc.id,
          name: data.name || '',
          category: data.category || 'UMUM',
          price_sell: Number(data.price_sell) || 0,
          qty: Number(data.qty) || 0,
          isPublished: data.isPublished ? 1 : 0,
          imageUrl: data.imageUrl || null,
          imagePublicId: data.imagePublicId || null,
          updatedAt: data.updatedAt || new Date().toISOString(),
          deleted: 0
        })
      })
      if (remoteProducts.length > 0) await productRepo.bulkSave(remoteProducts)
      return await productRepo.getPublished()
    } catch (error) {
      if (error.name === 'AbortError' || error.message?.includes('AbortError')) {
        console.warn('[ProductService] AbortError ignored (Vite HMR)')
        return await productRepo.getPublished()
      }
      console.error('[ProductService] Error:', error)
      return await productRepo.getPublished()
    }
  }
}

import { doc, getDoc } from 'firebase/firestore'
import { db as firestoreDb } from './firebase.js'

export const qrisService = {
  async fetchRemoteSettings() {
    if (!navigator.onLine) return null
    try {
      const docRef = doc(firestoreDb, 'settings', 'qris')
      const docSnap = await getDoc(docRef)
      if (docSnap.exists()) {
        const data = docSnap.data()
        return {
          id: 'qris',
          qrisUrl: data.qrisUrl || '',
          merchantName: data.merchantName || 'WARUNG SINAR PAGI',
          nmid: data.nmid || '',
          updatedAt: data.updatedAt || new Date().toISOString()
        }
      }
    } catch (error) {
      console.error('[QrisService] Error:', error)
    }
    return null
  }
}

// src/services/cloudinaryService.js - Sinar Pagi Web
const CLOUD_NAME = import.meta.env.VITE_CLOUDINARY_CLOUD_NAME || 'vezqidog'
const UPLOAD_PRESET = import.meta.env.VITE_CLOUDINARY_PRESET || 'sinarpagi-products'

/**
 * Compress via Canvas 800px JPEG 70%
 */
export function compressImage(file) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = (event) => {
      const img = new Image()
      img.onload = () => {
        const canvas = document.createElement('canvas')
        const maxWidth = 800
        let width = img.width
        let height = img.height
        if (width > maxWidth) {
          height = Math.round((height * maxWidth) / width)
          width = maxWidth
        }
        canvas.width = width
        canvas.height = height
        const ctx = canvas.getContext('2d')
        ctx.drawImage(img, 0, 0, width, height)
        canvas.toBlob(
          (blob) => (blob ? resolve(blob) : reject(new Error('Gagal kompres'))),
          'image/jpeg',
          0.7
        )
      }
      img.onerror = (err) => reject(err)
      img.src = event.target.result
    }
    reader.onerror = (err) => reject(err)
    reader.readAsDataURL(file)
  })
}

/**
 * Upload ke Cloudinary (folder: sinar-pagi/products, sinar-pagi/qris, sinar-pagi/bukti-bayar)
 */
export async function uploadToCloudinary(imageBlob, folder = 'sinar-pagi/bukti-bayar') {
  const formData = new FormData()
  formData.append('file', imageBlob)
  formData.append('upload_preset', UPLOAD_PRESET)
  formData.append('folder', folder)

  const response = await fetch(`https://api.cloudinary.com/v1_1/${CLOUD_NAME}/image/upload`, {
    method: 'POST',
    body: formData
  })
  if (!response.ok) {
    const errorText = await response.text()
    throw new Error(`Cloudinary Upload Failed: ${errorText}`)
  }
  const data = await response.json()
  return {
    secure_url: data.secure_url,
    public_id: data.public_id
  }
}

export function getThumbUrl(publicIdOrUrl, width = 300) {
  if (!publicIdOrUrl) return ''
  if (!publicIdOrUrl.includes('res.cloudinary.com')) {
    return `https://res.cloudinary.com/${CLOUD_NAME}/image/upload/w_${width},f_auto,q_auto/${publicIdOrUrl}`
  }
  return publicIdOrUrl.replace('/upload/', `/upload/w_${width},f_auto,q_auto/`)
}

const WA_PHONE = '6289516477760'

export const waService = {
  buildOrderMessage(customerData, cartItems, totalPrice, paymentMethod, buktiUrl = '') {
    let message = `Halo Warung Sinar Pagi, saya mau pesan:\n\n`

    cartItems.forEach((item, index) => {
      message += `${index + 1}. ${item.name} (${item.qty}x) - Rp${(item.price_sell * item.qty).toLocaleString('id-ID')}\n`
    })

    message += `\n*Total Belanja: Rp${totalPrice.toLocaleString('id-ID')}*\n\n`
    message += `*Data Pemesan:*\n`
    message += `Nama: ${customerData.name}\n`
    message += `Alamat/Catatan: ${customerData.address}\n`
    message += `Metode Pembayaran: ${paymentMethod}\n`

    if (paymentMethod === 'QRIS' && buktiUrl) {
      message += `Bukti Bayar: ${buktiUrl}\n`
    }

    message += `\nMohon segera diproses ya, terima kasih!`
    return message
  },

  sendOrderToWhatsApp(customerData, cartItems, totalPrice, paymentMethod, buktiUrl = '') {
    const message = this.buildOrderMessage(customerData, cartItems, totalPrice, paymentMethod, buktiUrl)
    const encoded = encodeURIComponent(message)
    const url = `https://wa.me/${WA_PHONE}?text=${encoded}`
    window.open(url, '_blank')
  }
}

export function buildCartItem(product, selectedType, isBrewed) {
  let finalPrice = product.price_sell
  let finalUnit = product.unit || 'PCS'
  let customName = product.name
  let notes = []

  if (selectedType === 'pack' && product.packPrice > 0) {
    finalPrice = product.packPrice
    finalUnit = product.purchasePackName || 'PAKET'
    customName = `${product.name} (${product.purchasePackName || 'Paket'})`
  }

  if (isBrewed) {
    notes.push('Minta Diseduh')
  }

  return {
    ...product,
    price_sell: finalPrice,
    unit: finalUnit,
    name: customName,
    note: notes.join(', ')
  }
}

===== src/db/repositories/productRepository.js =====
import { db } from '../index.js'

export const productRepo = {
  getPublished: async () => {
    const list = await db.products.where('isPublished').equals(1).toArray()
    return list.filter(p => !p.deleted)
  },
  getById: async (id) => {
    const item = await db.products.get(id)
    return item && item.isPublished && !item.deleted ? item : null
  },
  getByCategory: async (category) => {
    if (!category || category === 'SEMUA') return productRepo.getPublished()
    const list = await db.products.where('category').equals(category).toArray()
    return list.filter(p => p.isPublished && !p.deleted)
  },
  bulkSave: async (products) => {
    if (!products.length) return
    await db.transaction('rw', db.products, async () => {
      for (const p of products) {
        await db.products.put({ ...p, synced: 1 })
      }
    })
  }
}

import { db } from '../index.js'

export const cartRepo = {
  async getAll() {
    return await db.cart.toArray()
  },
  async getByProductId(productId) {
    return await db.cart.where('productId').equals(productId).first()
  },
  async save(item) {
    return await db.cart.put(item)
  },
  async updateQty(productId, qty) {
    const item = await this.getByProductId(productId)
    if (item) {
      if (qty <= 0) {
        await this.remove(productId)
      } else {
        item.qty = qty
        item.updatedAt = new Date().toISOString()
        await db.cart.put(item)
      }
    }
  },
  async remove(productId) {
    await db.cart.where('productId').equals(productId).delete()
  },
  async clear() {
    await db.cart.clear()
  }
}
